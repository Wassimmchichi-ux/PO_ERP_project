3) PARSED PO WORKFLOW
---------------------

3.1) Upload Parsed PO JSON File

curl -X POST "$API_URL/purchase-orders/parsed/upload-json" \
  -H "accept: application/json" \
  -F "file=@$PARSED_PO_JSON;type=application/json"

# Optional: save the response and extract parsed_po_id.
curl -s -X POST "$API_URL/purchase-orders/parsed/upload-json" \
  -F "file=@$PARSED_PO_JSON;type=application/json" \
  | tee /tmp/parsed_po_upload_response.json

export PARSED_PO_ID=$(jq -r '.parsed_po_id' /tmp/parsed_po_upload_response.json)
echo "$PARSED_PO_ID"


3.2) Create Parsed PO From JSON Body

# Use this only if your frontend already has the parsed JSON object.
# For file testing, prefer /upload-json above.
curl -X POST "$API_URL/purchase-orders/parsed" \
  -H "accept: application/json" \
  -H "Content-Type: application/json" \
  -d '{
    "source_file_name": "2301416000.json",
    "parser_name": "external-po-parser",
    "parser_version": "1",
    "parsed_payload": {
      "po_number": "2301416000",
      "version": "1",
      "line_count": 1,
      "stated_total": 689.64,
      "computed_total": 689.64,
      "totals_match": true,
      "lines": [
        {
          "line_no": "00001",
          "plant": "7142",
          "due_date": "2025-11-01",
          "material_group": "907010",
          "part_number": "5618912-9L43-AL",
          "description": "RB40%LH COMF. ART. FAB.",
          "customer_part_ref": "A1749200600",
          "item_category": "Prototype Parts",
          "quantity": 21,
          "unit_price": 32.84,
          "total": 689.64
        }
      ]
    }
  }'


3.3) Get Parsed PO Review Data

curl -X GET "$API_URL/purchase-orders/parsed/$PARSED_PO_ID" \
  -H "accept: application/json"


3.4) Save Corrections

# Important:
# - Do not send the full parsed PO response here.
# - Send only correction diffs.
# - lines[0] means first JSON array item, not PO line number 00001.
# - Validation is automatically run after corrections are saved.
# - Response status becomes VALIDATED if clean, otherwise CORRECTED.
curl -X PUT "$API_URL/purchase-orders/parsed/$PARSED_PO_ID/corrections" \
  -H "accept: application/json" \
  -H "Content-Type: application/json" \
  -d '{
    "corrections": [
      {
        "field_path": "lines[0].quantity",
        "corrected_value": 22
      },
      {
        "field_path": "contact_email",
        "corrected_value": "launch@example.com"
      }
    ]
  }'


3.5) Validate Corrected Parsed PO

# Optional recheck. Corrections already trigger validation automatically.
curl -X POST "$API_URL/purchase-orders/parsed/$PARSED_PO_ID/validate" \
  -H "accept: application/json"


3.6) Import Corrected Parsed PO Into Real PO Tables

# This creates/updates:
# - PurchaseOrder
# - PurchaseOrderVersion
# - PurchaseOrderLine
# - FGPN
# - FGPNPriceHistory, if existing FGPN price changed
curl -X POST "$API_URL/purchase-orders/parsed/$PARSED_PO_ID/import" \
  -H "accept: application/json" \
  -H "Content-Type: application/json" \
  -d "{
    \"project_id\": $PROJECT_ID,
    \"customer_id\": $CUSTOMER_ID,
    \"contact_id\": $CONTACT_ID,
    \"delivery_method_id\": $DELIVERY_METHOD_ID,
    \"change_reason\": \"Imported from corrected parsed PO\"
  }"

# Optional: save import response.
curl -s -X POST "$API_URL/purchase-orders/parsed/$PARSED_PO_ID/import" \
  -H "Content-Type: application/json" \
  -d "{
    \"project_id\": $PROJECT_ID,
    \"customer_id\": $CUSTOMER_ID,
    \"contact_id\": $CONTACT_ID,
    \"delivery_method_id\": $DELIVERY_METHOD_ID,
    \"change_reason\": \"Imported from corrected parsed PO\"
  }" | tee /tmp/parsed_po_import_response.json

export PO_ID=$(jq -r '.po_id' /tmp/parsed_po_import_response.json)
echo "$PO_ID"

6) HELPFUL DATABASE LOOKUPS FOR IDS
-----------------------------------

# Run from repo root if you use docker-compose DB.
# These are not API endpoints, but they help fill PO_ID, FGPN_ID, PO_LINE_ID, etc.

docker compose exec -T db psql -U postgres -d po_database -c 'select project_id, project_code, name from "Project";'

docker compose exec -T db psql -U postgres -d po_database -c 'select customer_id, customer_code, name from "Customer";'

docker compose exec -T db psql -U postgres -d po_database -c 'select contact_id, customer_id, name from "Contact";'

docker compose exec -T db psql -U postgres -d po_database -c 'select delivery_method_id, name from "DeliveryMethod";'

docker compose exec -T db psql -U postgres -d po_database -c 'select po_id, po_number, current_version_id from "PurchaseOrder" order by po_id desc limit 10;'

docker compose exec -T db psql -U postgres -d po_database -c 'select fgpn_id, project_id, fgpn_code, unit_price from "FGPN" order by fgpn_id desc limit 20;'

docker compose exec -T db psql -U postgres -d po_database -c 'select po_line_id, po_version_id, line_number, fgpn_id, ordered_quantity from "PurchaseOrderLine" order by po_line_id desc limit 20;'

docker compose exec -T db psql -U postgres -d po_database -c 'select po.po_id, po.po_number, pol.po_line_id, f.fgpn_id, f.fgpn_code, pol.ordered_quantity from "PurchaseOrder" po join "PurchaseOrderVersion" pov on pov.po_id = po.po_id and pov.po_version_id = po.current_version_id join "PurchaseOrderLine" pol on pol.po_version_id = pov.po_version_id join "FGPN" f on f.fgpn_id = pol.fgpn_id order by po.po_id desc, pol.line_number limit 30;'

docker compose exec -T db psql -U postgres -d po_database -c 'select document_id, document_type, file_name from "Document" order by document_id desc limit 20;'

docker compose exec -T db psql -U postgres -d po_database -c 'select id, full_name, role from "users" where role in ('\''LAUNCH ENGINEER'\'', '\''LAUNCH MANAGER'\'') order by full_name;'

docker compose exec -T db psql -U postgres -d po_database -c 'select stock_unit_id, material_part_number, quantity, status, user_id from "StockUnit" order by stock_unit_id desc limit 20;'
